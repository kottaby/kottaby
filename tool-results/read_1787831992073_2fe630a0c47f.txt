   102→// ─── Scroll-spy hook ───────────────────────────────────────────────
   103→
   104→function useScrollSpy(sectionIds: readonly string[]): string {
   105→  const [activeId, setActiveId] = useState("");
   106→  useEffect(() => {
   107→    const observers: IntersectionObserver[] = [];
   108→    for (const id of sectionIds) {
   109→      const el = document.getElementById(id);
   110→      if (!el) continue;
   111→      const obs = new IntersectionObserver(
   112→        ([entry]) => {
   113→          if (entry.isIntersecting) setActiveId(id);
   114→        },
   115→        { threshold: 0.2, rootMargin: "-80px 0px -50% 0px" }
   116→      );
   117→      obs.observe(el);
   118→      observers.push(obs);
   119→    }
   120→    return () => {
   121→      for (const o of observers) o.disconnect();
   122→    };
   123→  }, [sectionIds]);
   124→  return activeId;
   125→}

   622→const isNumericChar = (ch: string): boolean => (ch >= "0" && ch <= "9") || ch === ",";

   625→/** RFC-lite email-shape check for client-side hints: single "@", non-empty
   626→ * local and domain parts, at least one dot in the domain, no whitespace and
   627→ * no further "@" characters. Linear scan — avoids the regex-backtracking
   628→ * class (sonarjs/super-linear-regex) entirely. */
   629→function isEmailLike(value: string): boolean {
   630→  const at = value.indexOf("@");
   631→  if (at <= 0 || at === value.length - 1) return false;
   632→  const domain = value.slice(at + 1);
   633→  if (domain.includes("@") || /\s/.test(value)) return false;
   634→  const dot = domain.lastIndexOf(".");
   635→  return dot > 0 && dot < domain.length - 1;
   636→}

   749→function AnimatedCounter({ raw }: { readonly raw: string }): ReactNode {
   750→  const { num, suffix } = parseStatValue(raw);
   751→  const [count, setCount] = useState(0);
   752→  const [started, setStarted] = useState(false);
   753→  const spanRef = useRef<HTMLSpanElement>(null);
   754→
   755→  useEffect(() => {
   756→    const el = spanRef.current;
   757→    let observer: IntersectionObserver | undefined;
   758→    if (el) {
   759→      observer = new IntersectionObserver(
   760→        ([entry]) => {
   761→          if (entry.isIntersecting && !started) {
   762→            setStarted(true);
   763→            observer?.unobserve(el);
   764→          }
   765→        },
   766→        { threshold: 0.3 }
   767→      );
   768→      observer.observe(el);
   769→    }
   770→    return () => observer?.disconnect();
   771→  }, [started]);
   772→
   773→  useEffect(() => {
   774→    if (!started) return;
   775→    const duration = 2000;
   776→    const startTime = performance.now();
   777→
   778→    function easeOutCubic(x: number): number {
   779→      return 1 - (1 - x) ** 3;
   780→    }
   781→
   782→    function tick(now: number) {
   783→      const elapsed = now - startTime;
   784→      const progress = Math.min(elapsed / duration, 1);
   785→      const eased = easeOutCubic(progress);
   786→      setCount(Math.floor(eased * num));
   787→      if (progress < 1) {
   788→        requestAnimationFrame(tick);
   789→      } else {
   790→        setCount(num);
   791→      }
   792→    }
   793→
   794→    requestAnimationFrame(tick);
   795→  }, [started, num]);
   796→
   797→  return (
   798→    <span ref={spanRef}>
   799→      {count.toLocaleString()}
   800→      {suffix}
   801→    </span>
   802→  );
   803→}

   805→/** Parse a stat string like "120+" or "8,500+" into { num, suffix }.
   806→ * Linear scan instead of regex — avoids super-linear backtracking. */
   807→function parseStatValue(raw: string): { num: number; suffix: string } {
   808→  if (raw.length === 0 || !isDigitChar(raw[0])) return { num: 0, suffix: raw };
   809→  let end = 1;
   810→  while (end < raw.length && isNumericChar(raw[end])) {
   811→    end += 1;
   812→  }
   813→  const numPart = raw.slice(0, end);
   814→  const suffixPart = raw.slice(end);
   815→  return { num: parseInt(numPart.replace(/,/g, ""), 10) || 0, suffix: suffixPart };
   816→}

   865→// ─── Hijri date & prayer times (Cairo) ──────────────────────────────
   866→// Pure client-side solar astronomy — no external API. Standard
   867→// PrayTimes-style formulas: Julian day → sun declination + equation of
   868→// time; Egyptian General Authority angles (Fajr 19.5°, Isha 17.5°);
   869→// Shafi'i Asr (shadow factor 1). Cairo stays on DST, so the UTC offset
   870→// is derived per-instant from Intl. Rendered post-mount only to keep
   871→// SSR and client markup identical (time-dependent content).
   872→
   873→const CAIRO_LAT = 30.0444;
   874→const CAIRO_LNG = 31.2357;
   875→const FAJR_ANGLE = 19.5;
   876→const ISHA_ANGLE = 17.5;
   877→const SUNRISE_ALT_DEG = 0.833; // refraction-corrected solar disc
   878→
   879→const toRad = (deg: number): number => (deg * Math.PI) / 180;
   880→const toDeg = (rad: number): number => (rad * 180) / Math.PI;
   881→const fixHour = (h: number): number => ((h % 24) + 24) % 24;
   882→
   883→/** Sun declination (deg) and equation of time (hours) for a Julian day. */
   884→function sunPosition(jd: number): { declination: number; equationOfTime: number } {
   885→  const d = jd - 2451545.0;
   886→  const g = (((357.529 + 0.98560028 * d) % 360) + 360) % 360;
   887→  const q = (((280.459 + 0.98564736 * d) % 360) + 360) % 360;
   888→  const l = (((q + 1.915 * Math.sin(toRad(g)) + 0.02 * Math.sin(toRad(2 * g))) % 360) + 360) % 360;
   889→  const e = 23.439 - 0.00000036 * d;
   890→  const ra = toDeg(Math.atan2(Math.cos(toRad(e)) * Math.sin(toRad(l)), Math.cos(toRad(l)))) / 15;
   891→  const declination = toDeg(Math.asin(Math.sin(toRad(e)) * Math.sin(toRad(l))));
   892→  const equationOfTime = q / 15 - fixHour(ra);
   893→  return { declination, equationOfTime };
   894→}
   895→
   896→/** Hour angle (hours from local noon) for the sun at `angleDeg` below horizon. */
   897→function hourAngleFor(declination: number, angleDeg: number): number {
   898→  const cosH =
   899→    (-Math.sin(toRad(angleDeg)) - Math.sin(toRad(declination)) * Math.sin(toRad(CAIRO_LAT))) /
   900→    (Math.cos(toRad(declination)) * Math.cos(toRad(CAIRO_LAT)));
   901→  return toDeg(Math.acos(Math.min(1, Math.max(-1, cosH)))) / 15;
   902→}
   903→
   904→/** DST-aware UTC offset (hours) for Africa/Cairo at the given instant. */
   905→function cairoOffsetHours(date: Date): number {
   906→  try {
   907→    const parts = new Intl.DateTimeFormat("en-US", {
   908→      timeZone: "Africa/Cairo",
   909→      timeZoneName: "longOffset",
   910→    }).formatToParts(date);
   911→    const tz = parts.find(p => p.type === "timeZoneName")?.value ?? "GMT+03:00";
   912→    const m = /GMT([+-])(\d{2}):(\d{2})/.exec(tz);
   913→    if (!m) {
   914→      return 3;
   915→    }
   916→    const sign = m[1] === "-" ? -1 : 1;
   917→    return sign * (Number(m[2]) + Number(m[3]) / 60);
   918→  } catch {
   919→    return 3;
   920→  }
   921→}

   923→interface PrayerDaySchedule {
   924→  fajr: number;
   925→  sunrise: number;
   926→  dhuhr: number;
   927→  asr: number;
   928→  maghrib: number;
   929→  isha: number;
   930→}
   931→
   932→/** Cairo prayer times as local clock hours (0-24) for the date's solar day. */
   933→function cairoPrayerSchedule(date: Date): PrayerDaySchedule {
   934→  const offset = cairoOffsetHours(date);
   935→  const jd = date.getTime() / 86400000 + 2440587.5 - CAIRO_LNG / (15 * 24);
   936→  const { declination, equationOfTime } = sunPosition(jd);
   937→  const dhuhr = 12 + offset - CAIRO_LNG / 15 - equationOfTime;
   938→  const asrAltitude = toDeg(Math.atan(1 / (1 + Math.tan(toRad(Math.abs(CAIRO_LAT - declination))))));
   939→  return {
   940→    fajr: dhuhr - hourAngleFor(declination, FAJR_ANGLE),
   941→    sunrise: dhuhr - hourAngleFor(declination, SUNRISE_ALT_DEG),
   942→    dhuhr,
   943→    asr: dhuhr + hourAngleFor(declination, -asrAltitude),
   944→    maghrib: dhuhr + hourAngleFor(declination, SUNRISE_ALT_DEG),
   945→    isha: dhuhr + hourAngleFor(declination, ISHA_ANGLE),
   946→  };
   947→}

   949→/** Next prayer (skipping sunrise) as [key, hours-from-now-including-rollover]. */
   950→function nextPrayer(schedule: PrayerDaySchedule, nowHours: number): { key: string; inHours: number } {
   951→  const order: { key: string; at: number }[] = [
   952→    { key: "fajr", at: schedule.fajr },
   953→    { key: "dhuhr", at: schedule.dhuhr },
   954→    { key: "asr", at: schedule.asr },
   955→    { key: "maghrib", at: schedule.maghrib },
   956→    { key: "isha", at: schedule.isha },
   957→  ];
   958→  for (const p of order) {
   959→    if (p.at > nowHours) {
   960→      return { key: p.key, inHours: p.at - nowHours };
   961→    }
   962→  }
   963→  return { key: "fajr", inHours: schedule.fajr + 24 - nowHours };
   964→}

   966→/** UTC-midnight epoch ms for the date part of `d` (module scope: the React
   967→    compiler misreads the uppercase `Date.UTC` member as a component). */
   968→function utcMidnightOf(d: Date): number {
   969→  const t = new Date(d);
   970→  t.setUTCHours(0, 0, 0, 0);
   971→  return t.getTime();
   972→}
