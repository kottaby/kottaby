   102→   102→   102→// ─── Scroll-spy hook ───────────────────────────────────────────────

   104→   104→   104→function useScrollSpy(sectionIds: readonly string[]): string {
   105→   105→   105→  const [activeId, setActiveId] = useState("");
   106→   106→   106→  useEffect(() => {
   107→   107→   107→    const observers: IntersectionObserver[] = [];
   108→   108→   108→    for (const id of sectionIds) {
   109→   109→   109→      const el = document.getElementById(id);
   110→   110→   110→      if (!el) continue;
   111→   111→   111→      const obs = new IntersectionObserver(
   112→   112→   112→        ([entry]) => {
   113→   113→   113→          if (entry.isIntersecting) setActiveId(id);
   114→   114→   114→        },
   115→   115→   115→        { threshold: 0.2, rootMargin: "-80px 0px -50% 0px" }
   116→   116→   116→      );
   117→   117→   117→      obs.observe(el);
   118→   118→   118→      observers.push(obs);
   119→   119→   119→    }
   120→   120→   120→    return () => {
   121→   121→   121→      for (const o of observers) o.disconnect();
   122→   122→   122→    };
   123→   123→   123→  }, [sectionIds]);
   124→   124→   124→  return activeId;
   125→   125→   125→}
