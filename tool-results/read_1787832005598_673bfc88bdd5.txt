   102→   102→// ─── Scroll-spy hook ───────────────────────────────────────────────
   103→   103→
   104→   104→function useScrollSpy(sectionIds: readonly string[]): string {
   105→   105→  const [activeId, setActiveId] = useState("");
   106→   106→  useEffect(() => {
   107→   107→    const observers: IntersectionObserver[] = [];
   108→   108→    for (const id of sectionIds) {
   109→   109→      const el = document.getElementById(id);
   110→   110→      if (!el) continue;
   111→   111→      const obs = new IntersectionObserver(
   112→   112→        ([entry]) => {
   113→   113→          if (entry.isIntersecting) setActiveId(id);
   114→   114→        },
   115→   115→        { threshold: 0.2, rootMargin: "-80px 0px -50% 0px" }
   116→   116→      );
   117→   117→      obs.observe(el);
   118→   118→      observers.push(obs);
   119→   119→    }
   120→   120→    return () => {
   121→   121→      for (const o of observers) o.disconnect();
   122→   122→    };
   123→   123→  }, [sectionIds]);
   124→   124→  return activeId;
   125→   125→}

   614→   614→/** This layout makes <body> the scroll container (overflow: hidden auto), so
   615→   615→ * window scrollY stays 0 while the page visually scrolls. Reads take the
   616→   616→ * maximal position across all candidate scrollers; the capture-phase
   617→   617→ * listener catches the non-bubbling scroll event fired at the body. */
   618→   618→function getMaxScrollTop(): number {
   619→   619→  return Math.max(window.scrollY, document.documentElement.scrollTop, document.body.scrollTop);
   620→   620→}

   622→   622→const isNumericChar = (ch: string): boolean => (ch >= "0" && ch <= "9") || ch === ",";
   623→   623→const isDigitChar = (ch: string): boolean => ch >= "0" && ch <= "9";
   624→   624→
   625→   625→/** RFC-lite email-shape check for client-side hints: single "@", non-empty
   626→   626→ * local and domain parts, at least one dot in the domain, no whitespace and
   627→   627→ * no further "@" characters. Linear scan — avoids the regex-backtracking
   628→   628→ * class (sonarjs/super-linear-regex) entirely. */
   629→   629→function isEmailLike(value: string): boolean {
   630→   630→  const at = value.indexOf("@");
   631→   631→  if (at <= 0 || at === value.length - 1) return false;
   632→   632→  const domain = value.slice(at + 1);
   633→   633→  if (domain.includes("@") || /\s/.test(value)) return false;
   634→   634→  const dot = domain.lastIndexOf(".");
   635→   635→  return dot > 0 && dot < domain.length - 1;
   636→   636→}

   749→   749→function AnimatedCounter({ raw }: { readonly raw: string }): ReactNode {
   750→   750→  const { num, suffix } = parseStatValue(raw);
   751→   751→  const [count, setCount] = useState(0);
   752→   752→  const [started, setStarted] = useState(false);
   753→   753→  const spanRef = useRef<HTMLSpanElement>(null);
   754→   754→
   755→   755→  useEffect(() => {
   756→   756→    const el = spanRef.current;
   757→   757→    let observer: IntersectionObserver | undefined;
   758→   758→    if (el) {
   759→   759→      observer = new IntersectionObserver(
   760→   760→        ([entry]) => {
   761→   761→          if (entry.isIntersecting && !started) {
   762→   762→            setStarted(true);
   763→   763→            observer?.unobserve(el);
   764→   764→          }
   765→   765→        },
   766→   766→        { threshold: 0.3 }
   767→   767→      );
   768→   768→      observer.observe(el);
   769→   769→    }
   770→   770→    return () => observer?.disconnect();
   771→   771→  }, [started]);
   772→   772→
   773→   773→  useEffect(() => {
   774→   774→    if (!started) return;
   775→   775→    const duration = 2000;
   776→   776→    const startTime = performance.now();
   777→   777→
   778→   778→    function easeOutCubic(x: number): number {
   779→   779→      return 1 - (1 - x) ** 3;
   780→   780→    }
   781→   781→
   782→   782→    function tick(now: number) {
   783→   783→      const elapsed = now - startTime;
   784→   784→      const progress = Math.min(elapsed / duration, 1);
   785→   785→      const eased = easeOutCubic(progress);
   786→   786→      setCount(Math.floor(eased * num));
   787→   787→      if (progress < 1) {
   788→   788→        requestAnimationFrame(tick);
   789→   789→      } else {
   790→   790→        setCount(num);
   791→   791→      }
   792→   792→    }
   793→   793→
   794→   794→    requestAnimationFrame(tick);
   795→   795→  }, [started, num]);
   796→   796→
   797→   797→  return (
   798→   798→    <span ref={spanRef}>
   799→   799→      {count.toLocaleString()}
   800→   800→      {suffix}
   801→   801→    </span>
   802→   802→  );
   803→   803→}
   804→   804→
   805→   805→/** Parse a stat string like "120+" or "8,500+" into { num, suffix }.
   806→   806→ * Linear scan instead of regex — avoids super-linear backtracking. */
   807→   807→function parseStatValue(raw: string): { num: number; suffix: string } {
   808→   808→  if (raw.length === 0 || !isDigitChar(raw[0])) return { num: 0, suffix: raw };
   809→   809→  let end = 1;
   810→   810→  while (end < raw.length && isNumericChar(raw[end])) {
   811→   811→    end += 1;
   812→   812→  }
   813→   813→  const numPart = raw.slice(0, end);
   814→   814→  const suffixPart = raw.slice(end);
   815→   815→  return { num: parseInt(numPart.replace(/,/g, ""), 10) || 0, suffix: suffixPart };
   816→   816→}

   865→   865→// ─── Hijri date & prayer times (Cairo) ──────────────────────────────
   866→   866→// Pure client-side solar astronomy — no external API. Standard
   867→   867→// PrayTimes-style formulas: Julian day → sun declination + equation of
   868→   868→// time; Egyptian General Authority angles (Fajr 19.5°, Isha 17.5°);
   869→   869→// Shafi'i Asr (shadow factor 1). Cairo stays on DST, so the UTC offset
   870→   870→// is derived per-instant from Intl. Rendered post-mount only to keep
   871→   871→// SSR and client markup identical (time-dependent content).
   872→   872→
   873→   873→const CAIRO_LAT = 30.0444;
   874→   874→const CAIRO_LNG = 31.2357;
   875→   875→const FAJR_ANGLE = 19.5;
   876→   876→const ISHA_ANGLE = 17.5;
   877→   877→const SUNRISE_ALT_DEG = 0.833; // refraction-corrected solar disc
   878→   878→
   879→   879→const toRad = (deg: number): number => (deg * Math.PI) / 180;
   880→   880→const toDeg = (rad: number): number => (rad * 180) / Math.PI;
   881→   881→const fixHour = (h: number): number => ((h % 24) + 24) % 24;
   882→   882→
   883→   883→/** Sun declination (deg) and equation of time (hours) for a Julian day. */
   884→   884→function sunPosition(jd: number): { declination: number; equationOfTime: number } {
   885→   885→  const d = jd - 2451545.0;
   886→   886→  const g = (((357.529 + 0.98560028 * d) % 360) + 360) % 360;
   887→   887→  const q = (((280.459 + 0.98564736 * d) % 360) + 360) % 360;
   888→   888→  const l = (((q + 1.915 * Math.sin(toRad(g)) + 0.02 * Math.sin(toRad(2 * g))) % 360) + 360) % 360;
   889→   889→  const e = 23.439 - 0.00000036 * d;
   890→   890→  const ra = toDeg(Math.atan2(Math.cos(toRad(e)) * Math.sin(toRad(l)), Math.cos(toRad(l)))) / 15;
   891→   891→  const declination = toDeg(Math.asin(Math.sin(toRad(e)) * Math.sin(toRad(l))));
   892→   892→  const equationOfTime = q / 15 - fixHour(ra);
   893→   893→  return { declination, equationOfTime };
   894→   894→}
   895→   895→
   896→   896→/** Hour angle (hours from local noon) for the sun at `angleDeg` below horizon. */
   897→   897→function hourAngleFor(declination: number, angleDeg: number): number {
   898→   898→  const cosH =
   899→   899→    (-Math.sin(toRad(angleDeg)) - Math.sin(toRad(declination)) * Math.sin(toRad(CAIRO_LAT))) /
   900→   900→    (Math.cos(toRad(declination)) * Math.cos(toRad(CAIRO_LAT)));
   901→   901→  return toDeg(Math.acos(Math.min(1, Math.max(-1, cosH)))) / 15;
   902→   902→}

   904→   904→/** DST-aware UTC offset (hours) for Africa/Cairo at the given instant. */
   905→   905→function cairoOffsetHours(date: Date): number {
   906→   906→  try {
   907→   907→    const parts = new Intl.DateTimeFormat("en-US", {
   908→   908→      timeZone: "Africa/Cairo",
   909→   909→      timeZoneName: "longOffset",
   910→   910→    }).formatToParts(date);
   911→   911→    const tz = parts.find(p => p.type === "timeZoneName")?.value ?? "GMT+03:00";
   912→   912→    const m = /GMT([+-])(\d{2}):(\d{2})/.exec(tz);
   913→   913→    if (!m) {
   914→   914→      return 3;
   915→   915→    }
   916→   916→    const sign = m[1] === "-" ? -1 : 1;
   917→   917→    return sign * (Number(m[2]) + Number(m[3]) / 60);
   918→   918→  } catch {
   919→   919→    return 3;
   920→   920→  }
   921→   921→}

   923→   923→interface PrayerDaySchedule {
   924→   924→  fajr: number;
   925→   925→  sunrise: number;
   926→   926→  dhuhr: number;
   927→   927→  asr: number;
   928→   928→  maghrib: number;
   929→   929→  isha: number;
   930→   930→}

   932→   932→/** Cairo prayer times as local clock hours (0-24) for the date's solar day. */
   933→   933→function cairoPrayerSchedule(date: Date): PrayerDaySchedule {
   934→   934→  const offset = cairoOffsetHours(date);
   935→   935→  const jd = date.getTime() / 86400000 + 2440587.5 - CAIRO_LNG / (15 * 24);
   936→   936→  const { declination, equationOfTime } = sunPosition(jd);
   937→   937→  const dhuhr = 12 + offset - CAIRO_LNG / 15 - equationOfTime;
   938→   938→  const asrAltitude = toDeg(Math.atan(1 / (1 + Math.tan(toRad(Math.abs(CAIRO_LAT - declination))))));
   939→   939→  return {
   940→   940→    fajr: dhuhr - hourAngleFor(declination, FAJR_ANGLE),
   941→   941→    sunrise: dhuhr - hourAngleFor(declination, SUNRISE_ALT_DEG),
   942→   942→    dhuhr,
   943→   943→    asr: dhuhr + hourAngleFor(declination, -asrAltitude),
   944→   944→    maghrib: dhuhr + hourAngleFor(declination, SUNRISE_ALT_DEG),
   945→   945→    isha: dhuhr + hourAngleFor(declination, ISHA_ANGLE),
   946→   946→  };
   947→   947→}

   952→   952→    { key: "fajr", at: schedule.fajr },

   954→   954→    { key: "asr", at: schedule.asr },
   955→   955→    { key: "maghrib", at: schedule.maghrib },
   956→   956→    { key: "isha", at: schedule.isha },

   958→   958→  for (const p of order) {
