   168→   168→function StudentsTab({
   169→   169→  students,
   170→   170→  onMutation,
   171→   171→}: {
   172→   172→  students: AdminStudent[] | null;
   173→   173→  onMutation: () => void;
   174→   174→}) {
   175→   175→  const { t } = useLocale();
   176→   176→  const [actingId, setActingId] = React.useState<string | null>(null);
   177→   177→  const [query, setQuery] = React.useState("");
   178→   178→  const [roleFilter, setRoleFilter] = React.useState<RoleFilter>("all");
   179→   179→  const [pageLimit, setPageLimit] = React.useState(20);
   180→   180→  const [selected, setSelected] = React.useState<Set<string>>(new Set());
   181→   181→
   182→   182→  if (students === null) return <LoadingState />;
   183→   183→
   184→   184→  // Client-side filter: by role + search query (name/email contains)
   185→   185→  const filtered = students.filter((s) => {
   186→   186→    if (roleFilter !== "all" && s.role !== roleFilter) return false;
   187→   187→    if (query.trim()) {
   188→   188→      const q = query.trim().toLowerCase();
   189→   189→      return (
   190→   190→        s.fullName.toLowerCase().includes(q) ||
   191→   191→        s.email.toLowerCase().includes(q)
   192→   192→      );
   193→   193→    }
   194→   194→    return true;
   195→   195→  });
   196→   196→
   197→   197→  if (students.length === 0) {
   198→   198→    return (
   199→   199→      <div className="space-y-3">
   200→   200→        <ExportButton type="students" />
   201→   201→        <EmptyState text={t.admin.noData} />
   202→   202→      </div>
   203→   203→    );
   204→   204→  }
   205→   205→
   206→   206→  const grantTrial = async (id: string) => {
   207→   207→    setActingId(id);
   208→   208→    try {
   209→   209→      const res = await fetch(`/api/admin/students/${id}`, {
   210→   210→        method: "PATCH",
   211→   211→        headers: { "Content-Type": "application/json" },
   212→   212→        body: JSON.stringify({ action: "grant-trial" }),
   213→   213→      });
   214→   214→      const data = await res.json();
   215→   215→      if (data.ok) {
   216→   216→        toast.success(t.admin.grantTrialSuccess);
   217→   217→        onMutation();
   218→   218→      } else {
   219→   219→        toast.error(data.error ?? "Grant failed");
   220→   220→      }
   221→   221→    } catch {
   222→   222→      toast.error("Network error");
   223→   223→    } finally {
   224→   224→      setActingId(null);
   225→   225→    }
   226→   226→  };
   227→   227→
   228→   228→  const deleteStudent = async (id: string) => {
   229→   229→    if (!window.confirm(t.admin.confirmDelete)) return;
   230→   230→    setActingId(id);
   231→   231→    try {
   232→   232→      const res = await fetch(`/api/admin/students/${id}`, { method: "DELETE" });
   233→   233→      const data = await res.json();
   234→   234→      if (data.ok) {
   235→   235→        toast.success(t.admin.deleted);
   236→   236→        onMutation();
   237→   237→      } else {
   238→   238→        toast.error(data.error ?? "Delete failed");
   239→   239→      }
   240→   240→    } catch {
   241→   241→      toast.error("Network error");
   242→   242→    } finally {
   243→   243→      setActingId(null);
   244→   244→    }
   245→   245→  };
   246→   246→
   247→   247→  const bulkDeleteStudents = async () => {
   248→   248→    if (selected.size === 0) return;
   249→   249→    if (!window.confirm(t.admin.confirmDelete)) return;
   250→   250→    let ok = 0;
   251→   251→    for (const id of selected) {
   252→   252→      try {
   253→   253→        const res = await fetch(`/api/admin/students/${id}`, { method: "DELETE" });
   254→   254→        const data = await res.json();
   255→   255→        if (data.ok) ok++;
   256→   256→      } catch {
   257→   257→        // skip
   258→   258→      }
   259→   259→    }
   260→   260→    toast.success(`${ok} ${t.admin.deleted.toLowerCase()}`);
   261→   261→    setSelected(new Set());
   262→   262→    onMutation();
   263→   263→  };
   264→   264→
   265→   265→  return (
   266→   266→    <div className="space-y-3">
   267→   267→      {/* Search + role filter + bulk + export */}
   268→   268→      <div className="flex flex-col sm:flex-row items-stretch sm:items-center gap-2 justify-between">
   269→   269→        <div className="relative flex-1 max-w-xs">
   270→   270→          <Search className="absolute start-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground pointer-events-none" />
   271→   271→          <Input
   272→   272→            value={query}
   273→   273→            onChange={(e) => setQuery(e.target.value)}
   274→   274→            placeholder={t.admin.search}
   275→   275→            className="h-8 ps-8 text-xs"
   276→   276→          />
   277→   277→        </div>
   278→   278→        <div className="flex items-center gap-1.5">
   279→   279→          {ROLE_FILTERS.map((r) => (
   280→   280→            <button
   281→   281→              key={r}
   282→   282→              type="button"
   283→   283→              onClick={() => setRoleFilter(r)}
   284→   284→              className={cn(
   285→   285→                "rounded-full px-2.5 py-0.5 text-[10px] font-semibold border transition-all",
   286→   286→                roleFilter === r
   287→   287→                  ? "bg-copper/15 border-copper/40 text-copper"
   288→   288→                  : "border-border text-muted-foreground hover:border-copper/30",
   289→   289→              )}
   290→   290→            >
   291→   291→              {r === "all" ? t.admin.overview : r === "student" ? t.admin.studentRole : r === "teacher" ? t.admin.teacherRole : t.admin.parentRole}
   292→   292→            </button>
   293→   293→          ))}
   294→   294→          {selected.size > 0 && (
   295→   295→            <button
   296→   296→              type="button"
   297→   297→              onClick={() => bulkDeleteStudents()}
   298→   298→              className="inline-flex items-center gap-1 rounded-md border border-destructive/30 bg-destructive/10 px-2 py-0.5 text-[10px] font-semibold text-destructive hover:bg-destructive/20"
   299→   299→            >
   300→   300→              <Trash2 className="h-3 w-3" />
   301→   301→              {t.admin.delete} ({selected.size})
   302→   302→            </button>
   303→   303→          )}
   304→   304→          <ExportButton type="students" />
   305→   305→        </div>
   306→   306→      </div>
   307→   307→      <div className="overflow-x-auto rounded-lg border border-border max-h-[40vh] overflow-y-auto">
   308→   308→        <table className="w-full text-xs">
   309→   309→          <thead className="sticky top-0 bg-card border-b border-border z-10">
   310→   310→            <tr>
   311→   311→              <th className="p-2 w-8">
   312→   312→                <input
   313→   313→                  type="checkbox"
   314→   314→                  checked={selected.size === filtered.length && filtered.length > 0}
   315→   315→                  onChange={() => {
   316→   316→                    if (selected.size === filtered.length) setSelected(new Set());
   317→   317→                    else setSelected(new Set(filtered.map((s) => s.id)));
   318→   318→                  }}
   319→   319→                  className="h-3.5 w-3.5 accent-copper"
   320→   320→                  aria-label="Select all"
   321→   321→                />
   322→   322→              </th>
   323→   323→              <th className="text-start p-2 font-semibold">{t.admin.fullName}</th>
   324→   324→              <th className="text-start p-2 font-semibold">{t.admin.email}</th>
   325→   325→              <th className="text-start p-2 font-semibold">{t.admin.role}</th>
   326→   326→              <th className="text-center p-2 font-semibold">{t.admin.trial}</th>
   327→   327→              <th className="text-center p-2 font-semibold">{t.admin.eligible}</th>
   328→   328→              <th className="text-center p-2 font-semibold">{t.admin.actions}</th>
   329→   329→            </tr>
   330→   330→          </thead>
   331→   331→          <tbody>
   332→   332→            {filtered.length === 0 ? (
   333→   333→              <tr>
   334→   334→                <td colSpan={7} className="p-8 text-center text-muted-foreground">
   335→   335→                  {t.admin.noData}
   336→   336→                </td>
   337→   337→              </tr>
   338→   338→            ) : (
   339→   339→              filtered.slice(0, pageLimit).map((s) => (
   340→   340→                <tr key={s.id} className="border-b border-border/50 hover:bg-muted/30">
   341→   341→                  <td className="p-2 text-center">
   342→   342→                    <input
   343→   343→                      type="checkbox"
   344→   344→                      checked={selected.has(s.id)}
   345→   345→                      onChange={() => {
   346→   346→                        const next = new Set(selected);
   347→   347→                        if (next.has(s.id)) next.delete(s.id);
   348→   348→                        else next.add(s.id);
   349→   349→                        setSelected(next);
   350→   350→                      }}
   351→   351→                      className="h-3.5 w-3.5 accent-copper"
   352→   352→                      aria-label={`Select ${s.fullName}`}
   353→   353→                    />
   354→   354→                  </td>
   355→   355→                  <td className="p-2 font-medium truncate max-w-[100px]">{s.fullName}</td>
   356→   356→                <td className="p-2 text-muted-foreground truncate max-w-[120px]">{s.email}</td>
   357→   357→                <td className="p-2">
   358→   358→                  <Badge variant="outline" className="text-[10px]">
   359→   359→                    {s.role === "student" ? t.admin.studentRole : s.role === "teacher" ? t.admin.teacherRole : t.admin.parentRole}
   360→   360→                  </Badge>
   361→   361→                </td>
   362→   362→                <td className="p-2 text-center">
   363→   363→                  {s.trialGrantedAt ? (
   364→   364→                    <span className="inline-flex items-center gap-0.5 text-copper">
   365→   365→                      <CheckCircle2 className="h-3.5 w-3.5" />
   366→   366→                      <span className="tabular-nums">{s.balanceTrial}</span>
   367→   367→                    </span>
   368→   368→                  ) : (
   369→   369→                    <XCircle className="h-3.5 w-3.5 text-muted-foreground inline" />
   370→   370→                  )}
   371→   371→                </td>
   372→   372→                <td className="p-2 text-center">
   373→   373→                  {s.eligible ? (
   374→   374→                    <Badge className="bg-green-500/15 text-green-500 border-0 text-[10px]">{t.admin.yes}</Badge>
   375→   375→                  ) : (
   376→   376→                    <Badge variant="outline" className="text-[10px]">{t.admin.no}</Badge>

   379→   379→                <td className="p-2 text-center">

   382→   382→                      <button
   383→   383→                        type="button"
   384→   384→                        onClick={() => grantTrial(s.id)}
   385→   385→                        disabled={actingId === s.id}
   386→   386→                        aria-label={t.admin.grantTrial}

   388→   388→                        className="inline-flex h-6 w-6 items-center justify-center rounded-md text-muted-foreground hover:text-copper hover:bg-copper/10 disabled:opacity-50"
   389→   389→                      >

   391→   391→                          <Loader2 className="h-3 w-3 animate-spin" />

   393→   393→                          <Gift className="h-3 w-3" />

   395→   395→                      </button>

   397→   397→                    <button

   400→   400→                      disabled={actingId === s.id}
   401→   401→                      aria-label={t.admin.deleteStudent}
   402→   402→                      title={t.admin.deleteStudent}
   403→   403→                      className="inline-flex h-6 w-6 items-center justify-center rounded-md text-muted-foreground hover:text-destructive hover:bg-destructive/10 disabled:opacity-50"
   404→   404→                    >
   405→   405→                      <Trash2 className="h-3 w-3" />
   406→   406→                    </button>
   407→   407→                  </div>
   408→   408→                </td>
   409→   409→              </tr>
   410→   410→              ))
   411→   411→            )}
   412→   412→          </tbody>
   413→   413→        </table>
   414→   414→      </div>
   415→   415→      {/* Filter count + load more */}
   416→   416→      <div className="flex items-center justify-between">
   417→   417→        <p className="text-[10px] text-muted-foreground">
   418→   418→          {Math.min(pageLimit, filtered.length)} / {filtered.length}
   419→   419→          {filtered.length !== students.length && ` (of ${students.length})`}
   420→   420→        </p>
   421→   421→        {pageLimit < filtered.length && (
   422→   422→          <button
   423→   423→            type="button"
   424→   424→            onClick={() => setPageLimit((v) => v + 20)}
   425→   425→            className="text-[10px] font-semibold text-copper hover:underline"
   426→   426→          >
   427→   427→            {t.admin.loadMore} →
   428→   428→          </button>
   429→   429→        )}
   430→   430→      </div>
   431→   431→    </div>
   432→   432→  );
   433→   433→}

   464→   464→  onMutation: () => void;
