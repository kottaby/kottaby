   168→function StudentsTab({
   169→  students,
   170→  onMutation,
   171→}: {
   172→  students: AdminStudent[] | null;
   173→  onMutation: () => void;
   174→}) {
   175→  const { t } = useLocale();
   176→  const [actingId, setActingId] = React.useState<string | null>(null);
   177→  const [query, setQuery] = React.useState("");
   178→  const [roleFilter, setRoleFilter] = React.useState<RoleFilter>("all");
   179→  const [pageLimit, setPageLimit] = React.useState(20);
   180→  const [selected, setSelected] = React.useState<Set<string>>(new Set());
   181→
   182→  if (students === null) return <LoadingState />;
   183→
   184→  // Client-side filter: by role + search query (name/email contains)
   185→  const filtered = students.filter((s) => {
   186→    if (roleFilter !== "all" && s.role !== roleFilter) return false;
   187→    if (query.trim()) {
   188→      const q = query.trim().toLowerCase();
   189→      return (
   190→        s.fullName.toLowerCase().includes(q) ||
   191→        s.email.toLowerCase().includes(q)
   192→      );
   193→    }
   194→    return true;
   195→  });
   196→
   197→  if (students.length === 0) {
   198→    return (
   199→      <div className="space-y-3">
   200→        <ExportButton type="students" />
   201→        <EmptyState text={t.admin.noData} />
   202→      </div>
   203→    );
   204→  }
   205→
   206→  const grantTrial = async (id: string) => {
   207→    setActingId(id);
   208→    try {
   209→      const res = await fetch(`/api/admin/students/${id}`, {
   210→        method: "PATCH",
   211→        headers: { "Content-Type": "application/json" },
   212→        body: JSON.stringify({ action: "grant-trial" }),
   213→      });
   214→      const data = await res.json();
   215→      if (data.ok) {
   216→        toast.success(t.admin.grantTrialSuccess);
   217→        onMutation();
   218→      } else {
   219→        toast.error(data.error ?? "Grant failed");
   220→      }
   221→    } catch {
   222→      toast.error("Network error");
   223→    } finally {
   224→      setActingId(null);
   225→    }
   226→  };
   227→
   228→  const deleteStudent = async (id: string) => {
   229→    if (!window.confirm(t.admin.confirmDelete)) return;
   230→    setActingId(id);
   231→    try {
   232→      const res = await fetch(`/api/admin/students/${id}`, { method: "DELETE" });
   233→      const data = await res.json();
   234→      if (data.ok) {
   235→        toast.success(t.admin.deleted);
   236→        onMutation();
   237→      } else {
   238→        toast.error(data.error ?? "Delete failed");
   239→      }
   240→    } catch {
   241→      toast.error("Network error");
   242→    } finally {
   243→      setActingId(null);
   244→    }
   245→  };
   246→
   247→  const bulkDeleteStudents = async () => {
   248→    if (selected.size === 0) return;
   249→    if (!window.confirm(t.admin.confirmDelete)) return;
   250→    let ok = 0;
   251→    for (const id of selected) {
   252→      try {
   253→        const res = await fetch(`/api/admin/students/${id}`, { method: "DELETE" });
   254→        const data = await res.json();
   255→        if (data.ok) ok++;
   256→      } catch {
   257→        // skip
   258→      }
   259→    }
   260→    toast.success(`${ok} ${t.admin.deleted.toLowerCase()}`);
   261→    setSelected(new Set());
   262→    onMutation();
   263→  };
   264→
   265→  return (
   266→    <div className="space-y-3">
   267→      {/* Search + role filter + bulk + export */}
   268→      <div className="flex flex-col sm:flex-row items-stretch sm:items-center gap-2 justify-between">
   269→        <div className="relative flex-1 max-w-xs">
   270→          <Search className="absolute start-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground pointer-events-none" />
   271→          <Input
   272→            value={query}
   273→            onChange={(e) => setQuery(e.target.value)}
   274→            placeholder={t.admin.search}
   275→            className="h-8 ps-8 text-xs"
   276→          />
   277→        </div>
   278→        <div className="flex items-center gap-1.5">
   279→          {ROLE_FILTERS.map((r) => (
   280→            <button
   281→              key={r}
   282→              type="button"
   283→              onClick={() => setRoleFilter(r)}
   284→              className={cn(
   285→                "rounded-full px-2.5 py-0.5 text-[10px] font-semibold border transition-all",
   286→                roleFilter === r
   287→                  ? "bg-copper/15 border-copper/40 text-copper"
   288→                  : "border-border text-muted-foreground hover:border-copper/30",
   289→              )}
   290→            >
   291→              {r === "all" ? t.admin.overview : r === "student" ? t.admin.studentRole : r === "teacher" ? t.admin.teacherRole : t.admin.parentRole}
   292→            </button>
   293→          ))}
   294→          {selected.size > 0 && (
   295→            <button
   296→              type="button"
   297→              onClick={() => bulkDeleteStudents()}
   298→              className="inline-flex items-center gap-1 rounded-md border border-destructive/30 bg-destructive/10 px-2 py-0.5 text-[10px] font-semibold text-destructive hover:bg-destructive/20"
   299→            >
   300→              <Trash2 className="h-3 w-3" />
   301→              {t.admin.delete} ({selected.size})
   302→            </button>
   303→          )}
   304→          <ExportButton type="students" />
   305→        </div>
   306→      </div>
   307→      <div className="overflow-x-auto rounded-lg border border-border max-h-[40vh] overflow-y-auto">
   308→        <table className="w-full text-xs">
   309→          <thead className="sticky top-0 bg-card border-b border-border z-10">
   310→            <tr>
   311→              <th className="p-2 w-8">
   312→                <input
   313→                  type="checkbox"
   314→                  checked={selected.size === filtered.length && filtered.length > 0}
   315→                  onChange={() => {
   316→                    if (selected.size === filtered.length) setSelected(new Set());
   317→                    else setSelected(new Set(filtered.map((s) => s.id)));
   318→                  }}
   319→                  className="h-3.5 w-3.5 accent-copper"
   320→                  aria-label="Select all"
   321→                />
   322→              </th>
   323→              <th className="text-start p-2 font-semibold">{t.admin.fullName}</th>
   324→              <th className="text-start p-2 font-semibold">{t.admin.email}</th>
   325→              <th className="text-start p-2 font-semibold">{t.admin.role}</th>
   326→              <th className="text-center p-2 font-semibold">{t.admin.trial}</th>
   327→              <th className="text-center p-2 font-semibold">{t.admin.eligible}</th>
   328→              <th className="text-center p-2 font-semibold">{t.admin.actions}</th>
   329→            </tr>
   330→          </thead>
   331→          <tbody>
   332→            {filtered.length === 0 ? (
   333→              <tr>
   334→                <td colSpan={7} className="p-8 text-center text-muted-foreground">
   335→                  {t.admin.noData}
   336→                </td>
   337→              </tr>
   338→            ) : (
   339→              filtered.slice(0, pageLimit).map((s) => (
   340→                <tr key={s.id} className="border-b border-border/50 hover:bg-muted/30">
   341→                  <td className="p-2 text-center">
   342→                    <input
   343→                      type="checkbox"
   344→                      checked={selected.has(s.id)}
   345→                      onChange={() => {
   346→                        const next = new Set(selected);
   347→                        if (next.has(s.id)) next.delete(s.id);
   348→                        else next.add(s.id);
   349→                        setSelected(next);
   350→                      }}
   351→                      className="h-3.5 w-3.5 accent-copper"
   352→                      aria-label={`Select ${s.fullName}`}
   353→                    />
   354→                  </td>
   355→                  <td className="p-2 font-medium truncate max-w-[100px]">{s.fullName}</td>
   356→                <td className="p-2 text-muted-foreground truncate max-w-[120px]">{s.email}</td>
   357→                <td className="p-2">
   358→                  <Badge variant="outline" className="text-[10px]">
   359→                    {s.role === "student" ? t.admin.studentRole : s.role === "teacher" ? t.admin.teacherRole : t.admin.parentRole}
   360→                  </Badge>
   361→                </td>
   362→                <td className="p-2 text-center">
   363→                  {s.trialGrantedAt ? (
   364→                    <span className="inline-flex items-center gap-0.5 text-copper">
   365→                      <CheckCircle2 className="h-3.5 w-3.5" />
   366→                      <span className="tabular-nums">{s.balanceTrial}</span>
   367→                    </span>
   368→                  ) : (
   369→                    <XCircle className="h-3.5 w-3.5 text-muted-foreground inline" />
   370→                  )}
   371→                </td>
   372→                <td className="p-2 text-center">
   373→                  {s.eligible ? (
   374→                    <Badge className="bg-green-500/15 text-green-500 border-0 text-[10px]">{t.admin.yes}</Badge>
   375→                  ) : (
   376→                    <Badge variant="outline" className="text-[10px]">{t.admin.no}</Badge>
   377→                  )}
   378→                </td>
   379→                <td className="p-2 text-center">
   380→                  <div className="inline-flex items-center gap-1">
   381→                    {!s.trialGrantedAt && (
   382→                      <button
   383→                        type="button"
   384→                        onClick={() => grantTrial(s.id)}
   385→                        disabled={actingId === s.id}
   386→                        aria-label={t.admin.grantTrial}
   387→                        title={t.admin.grantTrial}
   388→                        className="inline-flex h-6 w-6 items-center justify-center rounded-md text-muted-foreground hover:text-copper hover:bg-copper/10 disabled:opacity-50"
   389→                      >
   390→                        {actingId === s.id ? (
   391→                          <Loader2 className="h-3 w-3 animate-spin" />
   392→                        ) : (
   393→                          <Gift className="h-3 w-3" />
   394→                        )}
   395→                      </button>
   396→                    )}
   397→                    <button
   398→                      type="button"
   399→                      onClick={() => deleteStudent(s.id)}
   400→                      disabled={actingId === s.id}
   401→                      aria-label={t.admin.deleteStudent}
   402→                      title={t.admin.deleteStudent}
   403→                      className="inline-flex h-6 w-6 items-center justify-center rounded-md text-muted-foreground hover:text-destructive hover:bg-destructive/10 disabled:opacity-50"
   404→                    >
   405→                      <Trash2 className="h-3 w-3" />
   406→                    </button>
   407→                  </div>
   408→                </td>
   409→              </tr>
   410→              ))
   411→            )}
   412→          </tbody>
   413→        </table>
   414→      </div>
   415→      {/* Filter count + load more */}
   416→      <div className="flex items-center justify-between">
   417→        <p className="text-[10px] text-muted-foreground">
   418→          {Math.min(pageLimit, filtered.length)} / {filtered.length}
   419→          {filtered.length !== students.length && ` (of ${students.length})`}
   420→        </p>
   421→        {pageLimit < filtered.length && (
   422→          <button
   423→            type="button"
   424→            onClick={() => setPageLimit((v) => v + 20)}
   425→            className="text-[10px] font-semibold text-copper hover:underline"
   426→          >
   427→            {t.admin.loadMore} →
   428→          </button>
   429→        )}
   430→      </div>
   431→    </div>
   432→  );

   845→function EligibilityChecker() {
   846→  const { t } = useLocale();
   847→  const [studentId, setStudentId] = React.useState("");
   848→  const [loading, setLoading] = React.useState(false);
   849→  const [result, setResult] = React.useState<{
   850→    eligible: boolean;
   851→    hasTrial: boolean;
   852→    hasPaid: boolean;
   853→    balanceTrial: number;
   854→    student?: { fullName: string; email: string; role: string };
   855→  } | null>(null);
   856→
   857→  const check = async () => {
   858→    if (!studentId.trim()) return;
   859→    setLoading(true);
   860→    setResult(null);
   861→    try {
   862→      const res = await fetch(`/api/students/${studentId.trim()}`);
   863→      const data = await res.json();
   864→      if (data.ok) {
   865→        setResult({
   866→          eligible: data.eligibility.eligible,
   867→          hasTrial: data.eligibility.hasTrial,
   868→          hasPaid: data.eligibility.hasPaid,
   869→          balanceTrial: data.student.balanceTrial,
   870→          student: {
   871→            fullName: data.student.fullName,
   872→            email: data.student.email,
   873→            role: data.student.role,
   874→          },
   875→        });
   876→      } else {
   877→        setResult({ eligible: false, hasTrial: false, hasPaid: false, balanceTrial: 0 });
   878→      }
   879→    } catch {
   880→      setResult({ eligible: false, hasTrial: false, hasPaid: false, balanceTrial: 0 });
   881→    } finally {
   882→      setLoading(false);
   883→    }
   884→  };
   885→
   886→  return (
   887→    <div className="rounded-lg border border-copper/30 bg-copper/5 p-3 space-y-2">
   888→      <h4 className="text-xs font-semibold uppercase tracking-wider text-copper flex items-center gap-1.5">
   889→        <Search className="h-3.5 w-3.5" />
   890→        {t.admin.checkEligibility}
   891→      </h4>
   892→      <div className="flex gap-2">
   893→        <Input
   894→          value={studentId}
   895→          onChange={(e) => setStudentId(e.target.value)}
   896→          placeholder={t.admin.enterStudentId}
   897→          className="h-8 text-xs font-mono"
   898→          onKeyDown={(e) => e.key === "Enter" && check()}
   899→        />
   900→        <Button size="sm" onClick={check} disabled={loading} className="h-8 bg-copper text-copper-foreground hover:bg-copper/90 shrink-0">
   901→          {loading ? <Loader2 className="h-3 w-3 animate-spin" /> : t.admin.check}
   902→        </Button>
   903→      </div>
   904→      {result && (
   905→        <div className="text-xs space-y-1 pt-1">
   906→          {result.student ? (
   907→            <>
   908→              <p className="font-medium">{result.student.fullName} — {result.student.email}</p>
   909→              <p className="text-muted-foreground">
   910→                {t.admin.trial}: {result.balanceTrial} | {t.admin.eligible}:{" "}
   911→                <span className={result.eligible ? "text-green-500 font-medium" : "text-destructive"}>
   912→                  {result.eligible ? t.admin.yes : t.admin.no}
   913→                </span>
   914→              </p>
   915→            </>
   916→          ) : (
   917→            <p className="text-destructive">{t.admin.noData}</p>
   918→          )}
   919→        </div>
   920→      )}
   921→    </div>
   922→  );
   923→}

   927→function ExportButton({ type }: { type: "students" | "bookings" | "contacts" | "newsletter" }) {

   945→      toast.success(t.admin.export);
   946→    } catch {
   947→      toast.error("Export failed");
   948→    } finally {
   949→      setLoading(false);
