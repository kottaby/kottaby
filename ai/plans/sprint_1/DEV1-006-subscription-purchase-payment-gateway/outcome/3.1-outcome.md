# Task 3.1 Outcome — Billing types (new + extensions)

**Date:** 2026-09-07
**Agent:** Billing Types Subagent
**Requirements:** REQ-003, REQ-005, REQ-042, REQ-060

## Summary

All canonical billing types per plan §2.3 implemented in `backend/types/billing/`. Select/Insert types derive from the Drizzle tables via `$inferSelect`/`$inferInsert` (repo AGENTS.md derivation pattern — no hand-constructed shapes, no `toReturnType` helper exists in this repo; the established derivation is `typeof table.$inferSelect` plus `Omit<SelectType, ...> & { ... }` for enum re-typing, per `backend/types/AGENTS.md` and the `registration.types.ts` precedent). ReturnType shapes re-type the raw pgEnum string-literal unions to the canonical TS enums. Port types are provider-agnostic, money-as-string, no service-layer type files anywhere.

```ts
// subscription.types.ts
export type SubscriptionSelectType    = typeof subscriptions.$inferSelect;       // (pre-existing)
export type SubscriptionInsertType    = typeof subscriptions.$inferInsert;       // NEW (needed by 4.1 insertSubscription)
export type SubscriptionReturnType    = Omit<SubscriptionSelectType, "status" | "paymentMethod"> & {
  status: SubscriptionStatus;               // @/backend/enum/billing/subscription-status.enum
  paymentMethod: PaymentGateway | null;     // @/backend/enum/billing/payment-gateway.enum
};
export interface PurchaseSubscriptionInput  { readonly planId: number }            // sole client-supplied field (BOPLA)
export type PurchaseSubscriptionReturnType  = {
  readonly subscription: SubscriptionReturnType;
  readonly payment: StudentPaymentReturnType;
  readonly checkout: PaymentCheckoutSession;
};

// student-payment.types.ts
export type StudentPaymentSelectType  = typeof studentPayments.$inferSelect;     // (pre-existing)
export type StudentPaymentInsertType  = typeof studentPayments.$inferInsert;     // NEW (needed by 4.2 insertPayment)
export type StudentPaymentReturnType  = Omit<StudentPaymentSelectType, "status" | "paymentGateway"> & {
  status: PaymentStatus;                    // @/backend/enum/billing/payment-status.enum
  paymentGateway: PaymentGateway;
};

// subscription-purchase-idempotency.types.ts
export type SubscriptionPurchaseIdempotencySelectType = typeof subscriptionPurchaseIdempotency.$inferSelect;
export type SubscriptionPurchaseIdempotencyInsertType = typeof subscriptionPurchaseIdempotency.$inferInsert;

// payment-gateway.types.ts
export interface PaymentCheckoutInput {
  readonly studentId: number;               // provider customer metadata
  readonly planId: number;                  // plan identity metadata
  readonly amount: string;                  // decimal string, copied verbatim from plan (money-as-string)
  readonly currency: string;
}
export interface PaymentCheckoutSession {
  readonly provider: PaymentGateway;        // TS enum value ("mock" member available via 3.2)
  readonly providerReference: string;       // mock adapter issues mock_<uuid>
  readonly checkoutUrl: string | null;      // nullable — server-side/mock providers have none
}
export interface PaymentWebhookEvent {
  readonly reference: string;
  readonly outcome: "confirmed" | "failed"; // literal union per plan §2.3 (not a pgEnum mirror)
  readonly amount: string;                  // decimal string
  readonly currency: string;
}
export interface PaymentGatewayPort {
  createCheckout(input: PaymentCheckoutInput): Promise<PaymentCheckoutSession>;
  parseWebhookEvent(rawBody: string): PaymentWebhookEvent;
}
```
