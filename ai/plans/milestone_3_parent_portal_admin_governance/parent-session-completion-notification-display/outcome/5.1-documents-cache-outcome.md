1. **Hook usage pattern (the sanctioned consumption):**
   ```ts
   import { useQuery } from "@apollo/client/react";
   import { parentSessionTargetQueryDocument } from "@/frontend/graphql/sharedDocuments";
   // skip-guarded: only while ?session= is present
   const { data, error, loading } = useQuery(parentSessionTargetQueryDocument, {
     variables: { sessionId }, // Exact<{ sessionId: number }> — a NUMBER (Int!), never a string
   ```
